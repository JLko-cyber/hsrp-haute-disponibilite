! ============================================================
! INTERFACE LAN - GigabitEthernet0/0
! Connexion au réseau local 192.168.1.0/24
! ============================================================
interface GigabitEthernet0/0
 description LAN - Réseau local 192.168.1.0/24
 ip address 192.168.1.1 255.255.255.0
 duplex auto
 speed auto
 media-type rj45
 !
 ! --- Configuration HSRP Groupe 1 ---

 standby 1 ip 192.168.1.254
 standby 1 priority 110
 standby 1 preempt
 standby 1 timers 1 3
 standby 1 track GigabitEthernet0/1 10

 !
 no shutdown
!
! ============================================================
! INTERFACE WAN - GigabitEthernet0/1
! Connexion vers le réseau externe / Internet
! ============================================================
interface GigabitEthernet0/1
 description WAN - Connexion externe
 ip address 10.0.0.1 255.255.255.252
 duplex auto
 speed auto
 media-type rj45
 no shutdown
!
! ============================================================
! INTERFACE LIAISON INTER-ROUTEURS - GigabitEthernet0/2

 description Liaison inter-routeurs R1-R2
 ip address 172.16.0.1 255.255.255.252
 duplex auto
 speed auto
 media-type rj45
 no shutdown
!

! ROUTAGE
